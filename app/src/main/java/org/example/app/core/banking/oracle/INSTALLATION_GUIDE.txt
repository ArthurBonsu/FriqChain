🎯 STEP 7: INITIALIZE IN YOUR BLOCKCHAIN
─────────────────────────────────────────────────────────────────────

Add to your main blockchain initialization class:

import org.example.app.core.oracle.*;
import org.example.app.core.types.Address;
import java.math.BigDecimal;

public class BlockchainInitializer {
    
    public void initializeOracles() {
        // Create core managers
        OracleAggregator aggregator = new OracleAggregator();
        OracleStakingManager stakingManager = new OracleStakingManager();
        OracleReputationTracker reputationTracker = new OracleReputationTracker();
        
        // Create FX oracle
        Address fxOracleAddress = Address.fromHex("0x742d35Cc6634C0532925a3b844Bc9e7595f0bEb1");
        FXRateOracle fxOracle = new FXRateOracle(
            "FX_ORACLE_MAIN",
            fxOracleAddress,
            new BigDecimal("5000") // Minimum stake
        );
        
        // Stake tokens
        stakingManager.stake("FX_ORACLE_MAIN", "FX_RATE", new BigDecimal("10000"));
        fxOracle.addStake(new BigDecimal("10000"));
        
        // Register with aggregator
        aggregator.registerReporter(fxOracle);
        
        System.out.println("✓ Oracle system initialized successfully");
        System.out.println("  - FX Oracle: " + fxOracle.getReporterId());
        System.out.println("  - Staked amount: " + fxOracle.getStakedAmount());
        System.out.println("  - Status: " + (fxOracle.isActive() ? "ACTIVE" : "INACTIVE"));
    }
}

📖 STEP 8: INTEGRATE WITH CURRENCY EXCHANGE
─────────────────────────────────────────────────────────────────────

In CurrencyExchangeTransaction.java, add:

import org.example.app.core.oracle.FXRateOracle;
import org.example.app.core.oracle.OracleReporter;

public class CurrencyExchangeTransaction extends Transaction {
    
    private FXRateOracle fxOracle; // Add this field
    
    // Add setter
    public void setFxOracle(FXRateOracle oracle) {
        this.fxOracle = oracle;
    }
    
    // Use in your rate calculation
    public BigInteger calculateFairExchangeRate(String sourceToken, String targetToken) {
        try {
            String currencyPair = sourceToken + "/" + targetToken;
            OracleReporter.OracleData rateData = fxOracle.fetchData(currencyPair);
            
            BigDecimal rate = (BigDecimal) rateData.getValue();
            BigDecimal confidence = rateData.getConfidence();
            
            // Check confidence threshold
            if (confidence.compareTo(new BigDecimal("0.80")) < 0) {
                throw new Exception("Low confidence: " + confidence);
            }
            
            // Convert to BigInteger (18 decimal places)
            return rate.multiply(new BigDecimal("1000000000000000000")).toBigInteger();
            
        } catch (Exception e) {
            LOGGER.severe("Failed to fetch rate: " + e.getMessage());
            throw new RuntimeException("Exchange rate unavailable");
        }
    }
}
